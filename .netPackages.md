#### Install swagger package
    - dotnet add package Swashbuckle.AspNetCore
    - (Program.cs -> using Microsoft.OpenApi.Models;)
    - (Program.cs -> builder.Services.AddEndpointsApiExplorer();
    builder.Services.AddSwaggerGen();)
    - (Program.cs -> After var app = builder.Build(); -> app.UseSwagger();
    app.UseSwaggerUI();)

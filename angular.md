#### install latest stable version of angular
- npm install -g @angular/cli

#### install specific version of angular
- npm install -g @angular/cli@[versionNumber]

#### angular status
- ng version

#### create angular project
- ng new client

#### run angular code
- ng serve

#### ngx bootstrap
- npm install ngx-bootstrap@[12] bootstrap font-awesome

#### local Certificate
- brew install mkcert

#### create new local certificate
- sudo mkcert -install

#### create new localhost certificate and key file (first create ssl folder: mkdir ssl and cd ssl)
- sudo mkcert localhost
